`timescale 1ns / 1ps

module testbench;

    // Inputs
    reg clk;
    reg rst;
    reg btn_select;
    reg btn_add_money;
    reg btn_refill;

    // Outputs
    wire [3:0] an;
    wire [6:0] seg;
    wire       led_dispense;

    // Unit Under Test (UUT)
    coffee_machine uut (
        .clk(clk),
        .rst(rst),
        .btn_select(btn_select),
        .btn_add_money(btn_add_money),
        .btn_refill(btn_refill),
        .an(an),
        .seg(seg),
        .led_dispense(led_dispense)
    );

    // Clock Generation (100 MHz -> 10ns Period)
    always #5 clk = ~clk;

    // Helper Task for Debounced Button Press
    task press_button;
        input integer button_type; // 1: add_money, 2: select, 3: refill
        begin
            case (button_type)
                1: btn_add_money = 1;
                2: btn_select    = 1;
                3: btn_refill    = 1;
            endcase
            #100000; // Hold pulse across debounce sampling cycles
            btn_add_money = 0;
            btn_select    = 0;
            btn_refill    = 0;
            #100000;
        end
    endtask

    initial begin
        // System Initialization
        clk = 0;
        rst = 1;
        btn_select = 0;
        btn_add_money = 0;
        btn_refill = 0;

        #100;
        rst = 0; // Release Reset -> Enters IDLE State (Displays CUP3)
        #2000;

        // Transaction 1: Select Coffee & Add 5 Money Units
        press_button(2); // Press btn_select -> Moves to PAYMENT state ('P')
        
        repeat (5) begin
            press_button(1); // Insert 5 money units
        end
        // Automatically transitions to DISPENSE state ('d') -> Brews & decrements to CUP2

        #50000;
        $stop;
    end

endmodule
